+ 自顶向下（二维数组）：
#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int n; cin >> n;
      vector<vector<int>> v(n, vector<int>(n, 0));
      for(int i = 0; i < n; i ++) {
          for(int j = 0; j <= i; j ++) {
              cin >> v[i][j];
              if(i != 0) {
                  if (j == 0) v[i][j] += v[i - 1][j];
  		// 在三角形左边缘，无分叉，直接加上
                  else v[i][j] += max(v[i - 1][j - 1], v[i - 1][j]);
  		// 在三角形非左边缘，存在分叉或无分叉，取最大值加上。
  		// 右边缘无分叉也这样是因为这里取了最大值，
  		// 而三角形内元素总是在[0,100]区间内，可以不采取处理
              }
          }
      }
      int mx = 0;
      for(int &x : v[n - 1]) mx = max(mx, x);
      // 找出最大路径和
      cout << mx << endl;
      return 0;
  }
#+end_src

https://www.luogu.com.cn/problem/P1057
+ 二维数组：
#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int n, m; cin >> n >> m;
      vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
      // dp[i][j]表示当来到第i轮时，第j个人已经拿到球dp[i][j]次
      dp[0][1] = 1;
      // 初始化，开始时由第1号拿着球
      for(int i = 1; i <= m ; i ++) {
          for(int j = 1; j <= n; j ++) {
              dp[i][j] = dp[i - 1][j + 1 == n + 1 ? 1 : j + 1] +
                  dp[i - 1][j - 1 == 0 ? n : j - 1];
  	    // 状态转移：当前轮数这个人拿到球的传球方法
  	    // 等于右边传过来的方法加上左边传过来的方法，
  	    // 其中j + 1 == n + 1 ? 1 : j + 1表示向右传递，
  	    // j - 1 == 0 ? n : j - 1表示向左传递。
          }
      }
      cout << dp[m][1] << endl;
      // 最后一轮中第1号的传球方法数作为答案
      return 0;
  }
#+end_src

#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int t, m; cin >> t >> m;
      vector<int> dp(t + 1, 0);
      // 索引：采药时间；值：采药最大值
      for(int i = 0; i < m; i ++) {
          int w, v; cin >> w >> v;
          for(int j = t; j >= w; j --) {
              dp[j] = max(dp[j], dp[j - w] + v);
  	    // 如果能采到更大价值则换为采集更大价值的
  	    // 不能则保留原先选择
          }
      }
      cout << dp[t] << "\n";
      return 0;
  }

#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int n; cin >> n;
      char root = '*';
      unordered_map<char, pair<char, char>> mp;
      // 建立新二叉树
      for(int i = 0; i < n; i ++) {
          string tmp; cin >> tmp;
          if(root == '*') root = tmp[0];
  	// 设置根节点
          for(char &c : tmp) {
              if(c != '*' && !mp.contains(c))
                  mp[c] = make_pair('*', '*');
  	    // 遍历每一个合法节点并初始化
          }
          mp[tmp[0]].first = tmp[1], mp[tmp[0]].second = tmp[2];
  	// 连接父子节点
      }
      auto dfs = [&](auto &&self, char &c) -> string {
          if(c == '*') return "";
  	// 空节点跳出
          string s = "";
          s.push_back(c);
          return s + self(self, mp[c].first) + self(self, mp[c].second);
  	// 合并返回子节点、左分支结果、右分支结果
      };
      cout << dfs(dfs, root) << "\n";
      return 0;
  }

*** DONE P1706 全排列问题
https://www.luogu.com.cn/problem/P1706
#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int n; cin >> n;
      vector<bool> v(n, false);
      // 当前路径是否已访问过该元素
      vector<int> w;
      // DFS路径
      auto dfs = [&](auto&& self) {
          if(w.size() == n) {
              for(int &i : w) {
                  cout << "    " << i;
              }
              cout << "\n";
              return;
          }
  	// 遍历完所有元素，输出路径
          for(int i = 0; i < n; i ++) {
              if(v[i]) continue;
  	    // 已访问，跳过
              w.push_back(i + 1); v[i] = true;
  	    // 未访问，尾加数组并标记已访问
              self(self);
  	    // 加深度，继续访问
              w.pop_back(); v[i] = false;
  	    // 还原数组并取消标记，用于下一循环
          }
      };
      dfs(dfs);
      // 进入DFS
      return 0;
  }

*** DONE P1605 迷宫
https://www.luogu.com.cn/problem/P1605
#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      vector<vector<int>> dirs{
          {-1, 0}, {1, 0}, {0, 1}, {0, -1}
      };
      // 路径方向
      int n, m, t; cin >> n >> m >> t;
      int sx, sy, fx, fy; cin >> sx >> sy >> fx >> fy;
      vector<vector<bool>> visit(m + 1, vector<bool>(n + 1, false));
      // 经过路径
      for(int i = 0; i < t; i ++) {
          int x, y; cin >> x >> y;
          visit[y][x] = true;
      }
      // 为障碍物标注已经过，防止穿过
      int cnt = 0;
      auto dfs = [&](auto&& self, int x, int y) {
          if(x == fx && y == fy) {
              ++ cnt;
              return;
          }
  	// 到达终点，路径加1
          for(vector<int> &dir : dirs) {
              int cx = dir[1] + x, cy = dir[0] + y;
  	    // 计算移动后坐标
              if(1 <= cx && cx <= n && 1 <= cy && cy <= m) {
                  if(visit[cy][cx]) continue;
  		// 判断是否在范围内且是否经过
                  visit[cy][cx] = true;
                  self(self, cx, cy);
                  visit[cy][cx] = false;
  		// 标记并访问
              }
          }
      };
      visit[sy][sx] = true;
      // 防止重复经过起点
      dfs(dfs, sx, sy);
      cout << cnt << "\n";
      return 0;
  }

*** DONE P5318 【深基18.例3】查找文献
https://www.luogu.com.cn/problem/P5318
#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  vector<vector<int>> graph;
  // 邻接表存储图的边相关信息

  int main() {
      int n, m; cin >> n >> m;
      vector<vector<int>> v(m, vector<int>(2));
      for(vector<int> &w : v) cin >> w[0] >> w[1];
      sort(v.begin(), v.end());
      graph.resize(n + 1, vector<int>{});
      for(vector<int> &w : v) {
          graph[w[0]].push_back(w[1]);
      }
      // 读取边并排序插入邻接表
      vector<bool> visit(n + 1, false);
      auto dfs = [&](auto&& self, int t) -> void {
          cout << t << " ";
          for(int &elem : graph[t]) {
              if(visit[elem]) continue;
              visit[elem] = true;
              self(self, elem);
  	    // 访问后不标记回false：防止重复经过某节点
          }
          return;
      };
      visit[1] = true; dfs(dfs, 1);
      cout << "\n";
      // DFS前序
      for(int i = 1; i <= n; i ++) visit[i] = false;
      // 重置已访问
      queue<int> q;
      visit[1] = true; q.push(1);
      // 放入根节点，用于向下层序遍历
      while(q.size()) {
          int nw = q.front(); q.pop();
  	// 取出当前需遍历的父元素
          cout << nw << " ";
          for(int &elem : graph[nw]) {
              if(visit[elem]) continue;
              visit[elem] = true;
              q.push(elem);
  	    // 未访问的直接子元素放入队列，等待遍历
          }
      }
      cout << "\n";
      // BFS：利用队列进行层序遍历
      return 0;
  }
#+end_src
