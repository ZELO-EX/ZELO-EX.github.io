  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int n; cin >> n;
      vector<vector<int>> v(n, vector<int>(n, 0));
      for(int i = 0; i < n; i ++) {
          for(int j = 0; j <= i; j ++) {
              cin >> v[i][j];
              if(i != 0) {
                  if (j == 0) v[i][j] += v[i - 1][j];
  		// 在三角形左边缘，无分叉，直接加上
                  else v[i][j] += max(v[i - 1][j - 1], v[i - 1][j]);
  		// 在三角形非左边缘，存在分叉或无分叉，取最大值加上。
  		// 右边缘无分叉也这样是因为这里取了最大值，
  		// 而三角形内元素总是在[0,100]区间内，可以不采取处理
              }
          }
      }
      int mx = 0;
      for(int &x : v[n - 1]) mx = max(mx, x);
      // 找出最大路径和
      cout << mx << endl;
      return 0;
  }

  	    // j - 1 == 0 ? n : j - 1表示向左传递。
          }
      }

  	// 空节点跳出

          return s + self(self, mp[c].first) + self(self, mp[c].second);
  	// 合并返回子节点、左分支结果、右分支结果
      };
      cout << dfs(dfs, root) << "\n";
      return 0;
  }

  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      int n; cin >> n;
      vector<bool> v(n, false);
      // 当前路径是否已访问过该元素
      vector<int> w;
      // DFS路径
      auto dfs = [&](auto&& self) {
          if(w.size() == n) {
              for(int &i : w) {
                  cout << "    " << i;
              }
              cout << "\n";
              return;
          }
  	// 遍历完所有元素，输出路径
          for(int i = 0; i < n; i ++) {
              if(v[i]) continue;
  	    // 已访问，跳过
              w.push_back(i + 1); v[i] = true;
  	    // 未访问，尾加数组并标记已访问
              self(self);
  	    // 加深度，继续访问
              w.pop_back(); v[i] = false;
  	    // 还原数组并取消标记，用于下一循环
          }
      };
      dfs(dfs);
      // 进入DFS
      return 0;
  }

#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  int main() {
      vector<vector<int>> dirs{
          {-1, 0}, {1, 0}, {0, 1}, {0, -1}
      };
      // 路径方向
      int n, m, t; cin >> n >> m >> t;
      int sx, sy, fx, fy; cin >> sx >> sy >> fx >> fy;
      vector<vector<bool>> visit(m + 1, vector<bool>(n + 1, false));
      // 经过路径
      for(int i = 0; i < t; i ++) {
          int x, y; cin >> x >> y;
          visit[y][x] = true;
      }
      // 为障碍物标注已经过，防止穿过
      int cnt = 0;
      auto dfs = [&](auto&& self, int x, int y) {
          if(x == fx && y == fy) {
              ++ cnt;
              return;
          }
  	// 到达终点，路径加1
          for(vector<int> &dir : dirs) {
              int cx = dir[1] + x, cy = dir[0] + y;
  	    // 计算移动后坐标
              if(1 <= cx && cx <= n && 1 <= cy && cy <= m) {
                  if(visit[cy][cx]) continue;
  		// 判断是否在范围内且是否经过
                  visit[cy][cx] = true;
                  self(self, cx, cy);
                  visit[cy][cx] = false;
  		// 标记并访问
              }
          }
      };
      visit[sy][sx] = true;
      // 防止重复经过起点
      dfs(dfs, sx, sy);
      cout << cnt << "\n";
      return 0;
  }

#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  vector<vector<int>> graph;
  // 邻接表存储图的边相关信息

  int main() {
      int n, m; cin >> n >> m;
      vector<vector<int>> v(m, vector<int>(2));
      for(vector<int> &w : v) cin >> w[0] >> w[1];
      sort(v.begin(), v.end());
      graph.resize(n + 1, vector<int>{});
      for(vector<int> &w : v) {
          graph[w[0]].push_back(w[1]);
      }
      // 读取边并排序插入邻接表
      vector<bool> visit(n + 1, false);
      auto dfs = [&](auto&& self, int t) -> void {
          cout << t << " ";
          for(int &elem : graph[t]) {
              if(visit[elem]) continue;
              visit[elem] = true;
              self(self, elem);
  	    // 访问后不标记回false：防止重复经过某节点
          }
          return;
      };
      visit[1] = true; dfs(dfs, 1);
      cout << "\n";
      // DFS前序
      for(int i = 1; i <= n; i ++) visit[i] = false;
      // 重置已访问
      queue<int> q;
      visit[1] = true; q.push(1);
      // 放入根节点，用于向下层序遍历
      while(q.size()) {
          int nw = q.front(); q.pop();
  	// 取出当前需遍历的父元素
          cout << nw << " ";
          for(int &elem : graph[nw]) {
              if(visit[elem]) continue;
              visit[elem] = true;
              q.push(elem);
  	    // 未访问的直接子元素放入队列，等待遍历
          }
      }
      cout << "\n";
      // BFS：利用队列进行层序遍历
      return 0;
  }

  // 输入分别为 棋盘的高 棋盘的宽 马所在的纵坐标 马所在的横坐标

  	// 读取输入

  	// 初始化大小与内容
  	mp[y][x] = 0;

  	// 设置马的起点
  	while (!q.empty()) {

  		y = p[0], x = p[1];

      {0, -1}, {0, 1}
  };

  int main() {
      int n; cin >> n;
      vector<vci> mp(n, vci(n, 0));
      for(vci &s : mp) for(int &i : s) cin >> i;
      queue<vci> q;
      auto bfs = [&](int y, int x) {
          mp[y][x] = 2; q.push(vci{y, x});
          while(q.size()) {
              vci v = q.front(); q.pop();
              for(vci &dir : dirs) {
                  int cy = v[0] + dir[0], cx = v[1] + dir[1];
                  if(0 <= cy && cy < n && 0 <= cx && cx < n && !mp[cy][cx]) {
                      mp[cy][cx] = 2; q.push(vci{cy, cx});
                  }
              }
          }
  	// 为连接边界的0标记为2，方便后续操作
      };
      // 写好需要用的BFS Lambda
      for(int i = 0; i < n; i ++) {
          if(!mp[0][i]) bfs(0, i);
          if(!mp[n - 1][i]) bfs(n - 1, i);
      }
      for(int i = 0; i < n; i ++) {
          if(!mp[i][0]) bfs(i, 0);
          if(!mp[i][n - 1]) bfs(i, n - 1);
      }
      // 对在边界上的每一个0做一次BFS
      for(vci &v : mp) {
          for(int &i : v) {
              cout << 2 - i << " ";
  	    // 全部做2-i操作，1保持不变，
  	    // 原来在边界的“2”改为0，不在边界的“0”改为2
          }
          cout << "\n";
      }
      return 0;
  }
