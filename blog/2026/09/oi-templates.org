+ 纯函数写法

#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  const long long maxn = 2e5 + 5;
  long long fa[maxn], rk[maxn];

  void init() {
      for(long long i = 1; i <= maxn; i ++) {
          fa[i] = i, rk[i] = 1;
      }
  }

  long long find(long long x) {
      return fa[x] == x ? x : fa[x] = find(fa[x]);
  }

  void merge(long long x, long long y) {
      long long fx = find(x), fy = find(y);
      if(fx == fy) return;
      if(rk[fx] < rk[fy]) std::swap(fx, fy);
      fa[fx] = fy;
      if(rk[fx] == rk[fy]) rk[fx] ++;
  }

  // e.g. Luogu P1551
  using ll = long long;
  int main() {
      ios::sync_with_stdio(false);
      cin.tie(0); cout.tie(0);
      ll n, m, p; cin >> n >> m >> p;
      init();
      while(m --) {
          ll x, y; cin >> x >> y;
          merge(x, y);
      }
      while(p --) {
          ll x, y; cin >> x >> y;
          if(find(x) == find(y)) cout << "Yes\n";
          else cout << "No\n";
      }
      return 0;
  }
#+end_src

+ class写法

#+begin_src cpp
  #include <bits/stdc++.h>
  using namespace std;

  class DSU {
  private:
  	std::vector<long long> fa, rk;
  public:
  	explicit DSU(size_t size_) : fa(size_), rk(size_, 1) {
  		iota(fa.begin(), fa.end(), 0);
  	}

  	long long find(long long x) {
  		return fa[x] == x ? x : fa[x] = find(fa[x]);
  	}

  	void unite(long long x, long long y) {
  		long long fx = find(x), fy = find(y);
  		if(fx == fy) return;
  		if(rk[fx] < rk[fy]) std::swap(fx, fy);
  		fa[fx] = fy;
  		if(rk[fx] == rk[fy]) rk[fx] ++;
  	}
  };

  // e.g. Luogu P1551
  using ll = long long;
  int main() {
  	ios::sync_with_stdio(false);
  	cin.tie(0); cout.tie(0);
      ll n, m, p; cin >> n >> m >> p;
      DSU dsu(n + 1);
      while(m --) {
          ll x, y; cin >> x >> y;
          dsu.unite(x, y);
      }
      while(p --) {
          ll x, y; cin >> x >> y;
          if(dsu.find(x) == dsu.find(y)) cout << "Yes\n";
          else cout << "No\n";
      }
      return 0;
  }

#+end_src
